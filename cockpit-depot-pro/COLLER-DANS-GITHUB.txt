name: Fabriquer l'application

# Deux usages, volontairement séparés :
#  - pull_request     : compilation de VÉRIFICATION, sans keystore, sans secrets, sans APK publié.
#  - workflow_dispatch: compilation OFFICIELLE (lancée à la main, dépôt officiel et branche par défaut
#                       seulement), signée avec le keystore du projet, vérifiée avec apksigner,
#                       APK publié seulement si tout est conforme.
on:
  workflow_dispatch:
  pull_request:

permissions:
  contents: read

concurrency:
  group: build-${{ github.ref }}
  cancel-in-progress: true

env:
  # Empreinte SHA-256 du certificat qui signe les APK de Cockpit (information publique).
  # Si elle change, les mises à jour des utilisateurs ne fonctionneront plus : voir docs/SIGNATURE.md
  EXPECTED_SHA256: "3F:65:36:19:D1:04:80:04:54:55:1E:8B:18:26:5F:64:8E:37:5B:2E:1E:CB:8E:B5:12:BC:40:CD:8C:2E:6D:5D"

jobs:
  # ---------------------------------------------------------------------------
  # Contributions (pull requests) : vérifie que le projet compile.
  # Aucun APK n'est fabriqué ni publié : les tâches ci-dessous n'utilisent pas la signature,
  # donc ni le keystore officiel ni les futurs secrets ne sont concernés.
  # ---------------------------------------------------------------------------
  verification:
    name: Vérifier la contribution (sans signature)
    if: github.event_name == 'pull_request'
    runs-on: ubuntu-24.04
    timeout-minutes: 40
    permissions:
      contents: read
    steps:
      - name: Récupérer le code
        uses: actions/checkout@v4
        with:
          persist-credentials: false

      - name: Installer Java 17
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '17'

      - name: Installer le SDK Android
        uses: android-actions/setup-android@v3

      - name: Installer Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.7'

      - name: Compiler le code, les ressources et les fichiers de l'application
        shell: bash
        run: |
          set -o pipefail
          gradle :app:processDebugManifest :app:processDebugResources :app:compileDebugJavaWithJavac :app:mergeDebugAssets --no-daemon --stacktrace 2>&1 | tee build.log

      - name: Résumé
        if: always()
        shell: bash
        run: |
          {
            echo "## Vérification de la contribution"
            echo ""
            echo "Résultat de la compilation : **${{ job.status }}**"
            echo ""
            echo "Cette vérification ne signe rien et ne publie aucun APK. Seuls les mainteneurs fabriquent l'APK officiel."
          } >> "$GITHUB_STEP_SUMMARY"
          if [ "${{ job.status }}" != "success" ] && [ -f build.log ]; then
            ERR=$(grep -E "error:|FAILED|What went wrong|Execution failed" build.log | head -n 12 | cut -c1-250 | sed ':a;N;$!ba;s/\n/%0A/g')
            echo "::error title=La compilation a échoué::${ERR}"
          fi

      - name: Publier le journal
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: journal-verification
          path: build.log
          if-no-files-found: ignore

  # ---------------------------------------------------------------------------
  # Compilation officielle : uniquement à la main, uniquement dans le dépôt officiel
  # clutchjojoo-tech/cockpit et uniquement sur sa branche par défaut. Dans un fork, cette tâche est ignorée.
  # ---------------------------------------------------------------------------
  build:
    name: Compiler l'APK
    if: github.event_name == 'workflow_dispatch' && github.repository == 'clutchjojoo-tech/cockpit' && github.ref_name == github.event.repository.default_branch
    runs-on: ubuntu-24.04
    timeout-minutes: 40
    permissions:
      contents: read
    steps:
      - id: s1
        name: Récupérer le code
        uses: actions/checkout@v4
        continue-on-error: true

      - id: s2
        name: Installer Java 17
        uses: actions/setup-java@v5
        continue-on-error: true
        with:
          distribution: temurin
          java-version: '17'

      - id: s3
        name: Installer le SDK Android
        uses: android-actions/setup-android@v3
        continue-on-error: true

      - id: s4
        name: Installer Gradle
        uses: gradle/actions/setup-gradle@v4
        continue-on-error: true
        with:
          gradle-version: '8.7'

      - id: s5
        name: Fabriquer l'APK
        continue-on-error: true
        shell: bash
        run: |
          set -o pipefail
          gradle assembleDebug --no-daemon --stacktrace 2>&1 | tee build.log

      - id: s6
        name: Vérifier la signature (apksigner)
        if: steps.s5.outcome == 'success'
        continue-on-error: true
        shell: bash
        run: |
          set -o pipefail
          APK=app/build/outputs/apk/debug/app-debug.apk
          SDK="${ANDROID_HOME:-$ANDROID_SDK_ROOT}"
          find_signer() { ls "$SDK"/build-tools/*/apksigner 2>/dev/null | sort -V | tail -n 1; }
          APKSIGNER=$(find_signer)
          if [ -z "$APKSIGNER" ]; then
            echo "apksigner introuvable : installation des build-tools 34.0.0"
            yes | sdkmanager "build-tools;34.0.0" > /dev/null
            APKSIGNER=$(find_signer)
          fi
          if [ -z "$APKSIGNER" ]; then
            echo "::error title=apksigner introuvable::Impossible de vérifier la signature, l'APK n'est donc pas proposé au téléchargement."
            exit 1
          fi
          echo "Outil : $APKSIGNER"
          # 1) la signature de l'APK doit être valide
          if ! "$APKSIGNER" verify --verbose --print-certs "$APK" 2>&1 | tee sig.log; then
            echo "::error title=Signature invalide::apksigner verify a échoué."
            exit 1
          fi
          # 2) l'empreinte SHA-256 du certificat doit être celle de référence
          EXPECTED=$(echo "$EXPECTED_SHA256" | tr -d ':' | tr 'A-Z' 'a-z')
          GOTS=$(grep "certificate SHA-256 digest" sig.log | awk '{print $NF}' | tr 'A-Z' 'a-z')
          if [ -z "$GOTS" ]; then
            echo "::error title=Empreinte introuvable::apksigner n'a affiché aucune empreinte de certificat."
            exit 1
          fi
          echo "Empreinte attendue : $EXPECTED"
          for G in $GOTS; do
            echo "Empreinte de l'APK : $G"
            if [ "$G" != "$EXPECTED" ]; then
              echo "::error title=Signature différente::L'APK n'est pas signé avec la clé habituelle de Cockpit. Les mises à jour des utilisateurs seraient refusées par Android. L'APK n'est pas proposé au téléchargement. Voir docs/SIGNATURE.md"
              exit 1
            fi
          done
          echo "Signature conforme : les mises à jour fonctionneront."

      - name: Bilan
        if: always()
        shell: bash
        run: |
          set +e
          APK=app/build/outputs/apk/debug/app-debug.apk
          {
            echo "## Compilation de Cockpit"
            echo ""
            echo "| Étape | Résultat |"
            echo "|---|---|"
            echo "| Récupérer le code | ${{ steps.s1.outcome }} |"
            echo "| Java 17 | ${{ steps.s2.outcome }} |"
            echo "| SDK Android | ${{ steps.s3.outcome }} |"
            echo "| Gradle | ${{ steps.s4.outcome }} |"
            echo "| Fabriquer l'APK | ${{ steps.s5.outcome }} |"
            echo "| Signature (apksigner) | ${{ steps.s6.outcome }} |"
            echo ""
            if [ -f "$APK" ]; then
              VER=$(grep -m1 "versionName" app/build.gradle | sed "s/.*'\(.*\)'.*/\1/")
              SIZE=$(du -h "$APK" | cut -f1)
              SUM=$(sha256sum "$APK" | cut -d' ' -f1)
              echo "**Version :** $VER · **Taille :** $SIZE"
              echo ""
              echo "SHA-256 de l'APK : \`$SUM\`"
              echo ""
              if [ "${{ steps.s5.outcome }}" = "success" ] && [ "${{ steps.s6.outcome }}" = "success" ]; then
                echo "Télécharge l'artefact **cockpit-apk** en bas de cette page."
              else
                echo "L'APK n'est **pas** proposé au téléchargement : la compilation ou la vérification de signature a échoué. Le journal reste disponible dans l'artefact **journal-build**."
              fi
            fi
          } >> "$GITHUB_STEP_SUMMARY"
          echo "::notice title=Bilan des étapes::checkout=${{ steps.s1.outcome }} java=${{ steps.s2.outcome }} android=${{ steps.s3.outcome }} gradle=${{ steps.s4.outcome }} build=${{ steps.s5.outcome }} signature=${{ steps.s6.outcome }}"
          if [ "${{ steps.s5.outcome }}" != "success" ] && [ -f build.log ]; then
            ERR=$(grep -E "error:|FAILED|What went wrong|Execution failed|> " build.log | head -n 12 | cut -c1-250 | sed ':a;N;$!ba;s/\n/%0A/g')
            echo "::error title=La compilation a échoué::${ERR}"
            MSG=$(tail -n 30 build.log | cut -c1-250 | sed ':a;N;$!ba;s/\n/%0A/g')
            echo "::error title=Fin du journal Gradle::${MSG}"
          fi
          for o in "${{ steps.s1.outcome }}" "${{ steps.s2.outcome }}" "${{ steps.s3.outcome }}" "${{ steps.s4.outcome }}" "${{ steps.s5.outcome }}"; do
            [ "$o" = "success" ] || exit 1
          done
          [ "${{ steps.s6.outcome }}" = "success" ] || exit 1

      # Le journal est toujours disponible, même en cas d'échec.
      - name: Publier le journal
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: journal-build
          path: |
            build.log
            sig.log
          if-no-files-found: ignore

      # L'APK n'est proposé que si la compilation ET la vérification de signature ont réussi.
      - name: Publier l'APK
        uses: actions/upload-artifact@v4
        if: always() && steps.s5.outcome == 'success' && steps.s6.outcome == 'success'
        with:
          name: cockpit-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
